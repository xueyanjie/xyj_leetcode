# 回文链表

## 题目
https://leetcode.cn/problems/palindrome-linked-list/?envType=study-plan-v2&envId=top-100-liked

## 解题
思路：找到中间节点，反转链表后半部分，双指针比较
```go 
/**
 * Definition for singly-linked list.
 * type ListNode struct {
 *     Val int
 *     Next *ListNode
 * }
 */
func isPalindrome(head *ListNode) bool {
    l := getLength(head)
    if l == 0 || l == 1 {return true}
    midIndex := l / 2
    mid := head
    // 找到中间靠后的节点
    for i := 0; i < midIndex; i++ {
        mid = mid.Next 
    }
    // 反转后半段
    revHead := reverse(mid)
    a, b := head, revHead 
    for i := 0; i < midIndex; i++ {
        if a.Val != b.Val {
            return false 
        }
        a, b = a.Next, b.Next
    }
    return true 
}

// 获取链表长度
func getLength(head *ListNode) int {
    temp := head 
    length := 0
    for temp != nil {
        length++
        temp = temp.Next
    }
    return length
}

// 反转链表
func reverse(head *ListNode) *ListNode {
    var prev *ListNode 
    curr := head 
    for curr != nil {
        next := curr.Next 
        curr.Next = prev 
        prev = curr
        curr = next
    }
    return prev
}
```
