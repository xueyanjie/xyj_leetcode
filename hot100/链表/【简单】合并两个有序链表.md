# 合并两个有序链表

## 解题
注意后边的兜底
```golang 
/**
 * Definition for singly-linked list.
 * type ListNode struct {
 *     Val int
 *     Next *ListNode
 * }
 */
func mergeTwoLists(list1 *ListNode, list2 *ListNode) *ListNode {
    prehead := &ListNode{}
    temp := prehead
    for list1 != nil && list2 != nil {
        if list2 == nil || list1.Val <= list2.Val {
            temp.Next = list1 
            list1 = list1.Next  // 先赋值后移动，最后需要兜底
        } else {
            temp.Next = list2
            list2 = list2.Next  // 先赋值后移动，最后需要兜底 
        }
        temp = temp.Next 
    }
    if list1 != nil {
        temp.Next = list1 
    } else {
        temp.Next = list2
    }
    return prehead.Next 
}
```
