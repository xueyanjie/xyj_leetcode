# 最小栈

## 题目
https://leetcode.cn/problems/min-stack/description/?envType=study-plan-v2&envId=top-100-liked

## 解题
方法一：辅助栈
时间复杂度：O(1)，空间复杂度：O(n)
```go
 type MinStack struct {
    S []int   // 栈
    I []int   // 记录最小值index
}


func Constructor() MinStack {
    return MinStack{
        S: make([]int, 0),
        I: make([]int, 0),
    }
}


func (this *MinStack) Push(val int)  {
    this.S = append(this.S, val)  //入栈
    if len(this.I) == 0 {
        this.I = append(this.I, len(this.S) - 1) // 记录最小index
        return 
    }
    if val < this.S[this.I[len(this.I) - 1]] {
        this.I = append(this.I, len(this.S) - 1) // 记录最小index
    }
    return 
}


func (this *MinStack) Pop()  {
    // 栈顶是当前最小值则最小值记录也需要变更
    if this.I[len(this.I) - 1] == len(this.S) - 1 { 
        this.I = this.I[:len(this.I) - 1]
    }
    // pop
    this.S = this.S[:len(this.S) - 1]
    return 
}


func (this *MinStack) Top() int {
    if len(this.S) == 0 {
        return 0
    }
    return this.S[len(this.S) - 1]
}


func (this *MinStack) GetMin() int {
    return this.S[this.I[len(this.I) - 1]]
}

```
